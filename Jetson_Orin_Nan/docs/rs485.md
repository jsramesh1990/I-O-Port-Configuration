# RS485 Interface

## Overview

RS485 is an industrial standard for multi-point serial communication, supporting long distances and multiple devices on a single bus. The Jetson Orin Nano requires an external RS485 transceiver (e.g., MAX485, SP3485) connected to the UART pins.

## Hardware Setup

### Required Components
- RS485 transceiver (MAX485/SP3485/ADM485)
- 120Ω termination resistor
- Pull-up/pull-down resistors (optional)

## Electrical Specifications

| Parameter | Value |
|-----------|-------|
| Bus Voltage | -7V to +12V |
| Differential Voltage | ±1.5V to ±5V |
| Common Mode Range | -7V to +12V |
| Max Devices | 32 (standard), 256 (with load splitting) |
| Max Distance | 1200 meters |
| Max Speed | 10 Mbps (at 12m), 100 kbps (at 1200m) |
| Termination | 120Ω (at both ends) |
| Fail-safe Bias | 680Ω pull-up, 680Ω pull-down |

class RS485 {
private:
    int fd;
    std::string device;
    struct termios old_tio;
    bool auto_direction;
    GPIO* direction_pin;
    
    void setRTS(bool high) {
        int status;
        ioctl(fd, TIOCMGET, &status);
        if(high)
            status |= TIOCM_RTS;
        else
            status &= ~TIOCM_RTS;
        ioctl(fd, TIOCMSET, &status);
    }
    
    void delayMicroseconds(int us) {
        usleep(us);
    }
    
public:
    RS485(const std::string& dev, bool auto_dir = true) 
        : auto_direction(auto_dir), direction_pin(nullptr) {
        device = dev;
        if(!auto_direction) {
            direction_pin = new GPIO(36, GPIO::Direction::OUTPUT);
        }
    }
    
    ~RS485() {
        if(direction_pin) delete direction_pin;
    }
    
    bool open(int baudrate = 9600) {
        fd = ::open(device.c_str(), O_RDWR | O_NOCTTY);
        if(fd < 0) return false;
        
        tcgetattr(fd, &old_tio);
        struct termios tio = {0};
        
        tio.c_cflag = CS8 | CLOCAL | CREAD;
        tio.c_cflag |= (baudrate == 9600) ? B9600 : B115200;
        tio.c_iflag = 0;
        tio.c_oflag = 0;
        tio.c_lflag = 0;
        tio.c_cc[VMIN] = 0;
        tio.c_cc[VTIME] = 10;
        
        tcflush(fd, TCIFLUSH);
        tcsetattr(fd, TCSANOW, &tio);
        
        if(auto_direction) {
            enableAutoDirection();
        }
        
        return true;
    }
    
    void enableAutoDirection() {
        struct serial_rs485 rs485conf = {0};
        rs485conf.flags |= SER_RS485_ENABLED;
        rs485conf.flags |= SER_RS485_RTS_ON_SEND;
        rs485conf.flags &= ~SER_RS485_RTS_AFTER_SEND;
        rs485conf.delay_rts_before_send = 0;
        rs485conf.delay_rts_after_send = 0;
        ioctl(fd, TIOCSRS485, &rs485conf);
    }
    
    ssize_t write(const uint8_t* data, size_t len) {
        if(!auto_direction && direction_pin) {
            direction_pin->write(GPIO::Value::HIGH);
            delayMicroseconds(100);
        }
        
        ssize_t written = ::write(fd, data, len);
        tcdrain(fd);  // Wait for all data to be sent
        
        if(!auto_direction && direction_pin) {
            delayMicroseconds(100);
            direction_pin->write(GPIO::Value::LOW);
        }
        
        return written;
    }
    
    ssize_t read(uint8_t* buffer, size_t len, int timeout_ms = 100) {
        if(!auto_direction && direction_pin) {
            direction_pin->write(GPIO::Value::LOW);
        }
        
        fd_set set;
        struct timeval timeout;
        timeout.tv_sec = timeout_ms / 1000;
        timeout.tv_usec = (timeout_ms % 1000) * 1000;
        
        FD_ZERO(&set);
        FD_SET(fd, &set);
        
        int result = select(fd + 1, &set, NULL, NULL, &timeout);
        if(result > 0) {
            return ::read(fd, buffer, len);
        }
        
        return 0;
    }
    
    void close() {
        if(fd >= 0) {
            ::close(fd);
            fd = -1;
        }
    }
};

class ModbusRTUMaster {
    RS485 rs485;
    uint8_t slave_id;
    
    uint16_t calculateCRC(const uint8_t* data, size_t len) {
        uint16_t crc = 0xFFFF;
        for(size_t i = 0; i < len; i++) {
            crc ^= data[i];
            for(int j = 0; j < 8; j++) {
                if(crc & 0x0001) {
                    crc = (crc >> 1) ^ 0xA001;
                } else {
                    crc >>= 1;
                }
            }
        }
        return crc;
    }
    
public:
    ModbusRTUMaster(const std::string& device, int baudrate, uint8_t id) 
        : rs485(device, true), slave_id(id) {
        rs485.open(baudrate);
    }
    
    bool readHoldingRegisters(uint16_t start_addr, uint16_t count, uint16_t* values) {
        uint8_t frame[8];
        frame[0] = slave_id;
        frame[1] = 0x03;  // Read holding registers
        frame[2] = (start_addr >> 8) & 0xFF;
        frame[3] = start_addr & 0xFF;
        frame[4] = (count >> 8) & 0xFF;
        frame[5] = count & 0xFF;
        
        uint16_t crc = calculateCRC(frame, 6);
        frame[6] = crc & 0xFF;
        frame[7] = (crc >> 8) & 0xFF;
        
        rs485.write(frame, 8);
        
        uint8_t response[256];
        ssize_t len = rs485.read(response, sizeof(response), 100);
        
        if(len < 5) return false;
        
        // Verify CRC
        crc = calculateCRC(response, len - 2);
        if(crc != (response[len-2] | (response[len-1] << 8)))
            return false;
        
        // Extract values
        int data_len = response[2];
        for(int i = 0; i < count; i++) {
            values[i] = (response[3 + 2*i] << 8) | response[4 + 2*i];
        }
        
        return true;
    }
    
    bool writeSingleRegister(uint16_t address, uint16_t value) {
        uint8_t frame[8];
        frame[0] = slave_id;
        frame[1] = 0x06;  // Write single register
        frame[2] = (address >> 8) & 0xFF;
        frame[3] = address & 0xFF;
        frame[4] = (value >> 8) & 0xFF;
        frame[5] = value & 0xFF;
        
        uint16_t crc = calculateCRC(frame, 6);
        frame[6] = crc & 0xFF;
        frame[7] = (crc >> 8) & 0xFF;
        
        rs485.write(frame, 8);
        
        uint8_t response[8];
        ssize_t len = rs485.read(response, sizeof(response), 100);
        
        return len == 8 && memcmp(frame, response, 8) == 0;
    }
};

Multi-drop Network Implementation

class RS485Network {
    RS485 rs485;
    std::vector<uint8_t> slave_addresses;
    std::mutex bus_mutex;
    
    // Timing parameters
    const int TURNAROUND_DELAY = 100;  // microseconds
    const int RESPONSE_TIMEOUT = 100;  // milliseconds
    
public:
    RS485Network(const std::string& device, int baudrate) 
        : rs485(device, true) {
        rs485.open(baudrate);
    }
    
    bool sendToSlave(uint8_t slave_id, const uint8_t* data, size_t len) {
        std::lock_guard<std::mutex> lock(bus_mutex);
        
        // Build frame with slave ID
        uint8_t* frame = new uint8_t[len + 1];
        frame[0] = slave_id;
        memcpy(frame + 1, data, len);
        
        // Send data
        ssize_t sent = rs485.write(frame, len + 1);
        delete[] frame;
        
        return sent == (ssize_t)(len + 1);
    }
    
    int receiveFromSlave(uint8_t* buffer, size_t max_len) {
        std::lock_guard<std::mutex> lock(bus_mutex);
        return rs485.read(buffer, max_len, RESPONSE_TIMEOUT);
    }
    
    void scanNetwork() {
        slave_addresses.clear();
        
        for(uint8_t addr = 1; addr <= 247; addr++) {
            uint8_t test_frame[] = {0x00, 0x00};  // Function 0 - identity
            if(sendToSlave(addr, test_frame, 2)) {
                uint8_t response[16];
                if(rs485.read(response, sizeof(response), 50) > 0) {
                    slave_addresses.push_back(addr);
                    printf("Found device at address %d\n", addr);
                }
            }
            
            // Minimum delay between polls
            usleep(10000);
        }
    }
    
    void setSlaveTimeout(uint8_t slave_id, int timeout_ms) {
        // Implementation specific to slave device
        uint8_t command[] = {0x06, 0x00, 0x00, 
                             (timeout_ms >> 8) & 0xFF, 
                             timeout_ms & 0xFF};
        sendToSlave(slave_id, command, sizeof(command));
    }
};

Error Detection and Recovery
CRC and Error Handling
cpp

class RS485ErrorHandler {
    struct ErrorStats {
        uint32_t crc_errors = 0;
        uint32_t timeout_errors = 0;
        uint32_t framing_errors = 0;
        uint32_t retry_count = 0;
        uint32_t successful_transmissions = 0;
    } stats;
    
    uint16_t calculateCRC16(const uint8_t* data, size_t len) {
        uint16_t crc = 0xFFFF;
        for(size_t i = 0; i < len; i++) {
            crc ^= data[i];
            for(int j = 0; j < 8; j++) {
                if(crc & 0x0001) {
                    crc = (crc >> 1) ^ 0xA001;
                } else {
                    crc >>= 1;
                }
            }
        }
        return crc;
    }
    
public:
    bool transmitWithRetry(RS485& rs485, const uint8_t* data, 
                          size_t len, int max_retries = 3) {
        for(int retry = 0; retry < max_retries; retry++) {
            uint16_t crc = calculateCRC16(data, len - 2);
            ((uint8_t*)data)[len-2] = crc & 0xFF;
            ((uint8_t*)data)[len-1] = (crc >> 8) & 0xFF;
            
            rs485.write(data, len);
            
            uint8_t ack;
            if(rs485.read(&ack, 1, 50) == 1 && ack == 0x06) {
                stats.successful_transmissions++;
                return true;
            }
            
            stats.retry_count++;
            usleep(10000 * (1 << retry));  // Exponential backoff
        }
        
        stats.timeout_errors++;
        return false;
    }
    
    void printStats() {
        printf("RS485 Statistics:\n");
        printf("  Successful: %u\n", stats.successful_transmissions);
        printf("  CRC Errors: %u\n", stats.crc_errors);
        printf("  Timeouts: %u\n", stats.timeout_errors);
        printf("  Retries: %u\n", stats.retry_count);
    }
};

class RS485DMA {
    int uart_fd;
    int dma_fd;
    void* dma_buffer;
    size_t buffer_size;
    
    void setupDMA() {
        // Allocate DMA buffer
        dma_buffer = aligned_alloc(64, buffer_size);
        
        // Configure DMA channel for UART
        struct dma_slave_config config = {
            .direction = DMA_MEM_TO_DEV,
            .src_addr_width = DMA_SLAVE_BUSWIDTH_1_BYTE,
            .dst_addr_width = DMA_SLAVE_BUSWIDTH_1_BYTE,
            .src_maxburst = 16,
            .dst_maxburst = 16,
            .device_fc = false,
        };
        
        // Configure UART for DMA
        unsigned int cr;
        ioctl(uart_fd, TIOCMGET, &cr);
        cr |= (1 << 14);  // Enable DMA
        ioctl(uart_fd, TIOCMSET, &cr);
    }
    
public:
    void dmaTransmit(const uint8_t* data, size_t len) {
        memcpy(dma_buffer, data, len);
        
        // Start DMA transfer
        dma_async_tx_descriptor *tx = dma_prep_slave_sg(
            dma_chan, dma_buffer, len, DMA_MEM_TO_DEV, 0);
        
        dmaengine_submit(tx);
        dma_async_issue_pending(dma_chan);
        
        // Wait for completion
        dma_wait_for_async_tx(tx);
        
        // Small delay for RS485 turn-around
        usleep(100);
    }
};

Troubleshooting Guide
Common Issues

    No communication

        Check termination resistors (120Ω at both ends)

        Verify A/B lines are not swapped

        Check common ground connection

    Data corruption

        Reduce bus length or speed

        Add bias resistors (680Ω to VCC/GND)

        Check for reflections (proper termination)

    Intermittent failures

        Verify power supply to transceiver

        Check for electromagnetic interference

        Add ferrite beads on A/B lines

Best Practices

    Always terminate both ends of the bus with 120Ω resistors

    Use twisted pair cable (CAT5 or dedicated RS485 cable)

    Keep stub lengths short (< 1m) when connecting devices

    Implement fail-safe biasing for open/short circuit detection

    Add transient protection (TVS diodes) on A/B lines

    Use galvanic isolation for long runs or noisy environments

    Implement retry logic with exponential backoff

    Monitor bus errors to detect problems early
